{{- /* Renders the site logo as a . Context: a dict with an optional "scheme" key: (unset)/"light" : white logo only, source set with overbright AVIF preferred and the white PNG as fallback. "dark" : just the inverted black-on-transparent PNG. "auto" : prefers-color-scheme switches between the white (dark scheme) and inverted black (light scheme) logo. The white PNG is the only committed raster; the black variant is inverted at build time via images.Invert, so there's one asset to maintain and a corrupt/missing source fails the build. */ -}} {{- $scheme := "" -}} {{- if reflect.IsMap . -}}{{- $scheme = .scheme -}}{{- end -}} {{- $class := "nav-logo" -}} {{- if reflect.IsMap . -}}{{- with .class }}{{- $class = . -}}{{- end -}}{{- end -}} {{- $avif := resources.Get "images/ocno_icon.avif" -}} {{- $png := resources.Get "images/ocno_icon.png" -}} {{- $pngInverted := "" -}} {{- with $png }}{{- $pngInverted = .Filter images.Invert -}}{{- end -}} {{- if eq $scheme "dark" -}} {{- with $pngInverted }} {{- end }} {{- else if eq $scheme "auto" -}} {{- with $pngInverted }} {{- end }} {{- with $avif }} {{- end }} {{- with $png }} {{- end }} {{- else -}} {{- with $avif }} {{- end }} {{- with $png }} {{- end }} {{- end -}}
