{{- /* Calculate contrasting text color for accent color */ -}} {{- $accentColor := $.Site.Params.accentColor -}} {{- $accentContrastingColor := $.Site.Params.accentContrastingColor -}} {{- /* If accentContrastingColor is not set, calculate it automatically */ -}} {{- if not $accentContrastingColor -}} {{- /* Parse hex color to RGB */ -}} {{- $hex := replaceRE "#" "" (printf "%s" $accentColor) -}} {{- $r := substr $hex 0 2 -}} {{- $g := substr $hex 2 2 -}} {{- $b := substr $hex 4 2 -}} {{- $rInt := printf "0x%s" $r | int -}} {{- $gInt := printf "0x%s" $g | int -}} {{- $bInt := printf "0x%s" $b | int -}} {{- /* Calculate relative luminance */ -}} {{- $rNorm := div (float $rInt) 255.0 -}} {{- $gNorm := div (float $gInt) 255.0 -}} {{- $bNorm := div (float $bInt) 255.0 -}} {{- $rLum := cond (gt $rNorm 0.03928) (pow (div (add $rNorm 0.055) 1.055) 2.4) (div $rNorm 12.92) -}} {{- $gLum := cond (gt $gNorm 0.03928) (pow (div (add $gNorm 0.055) 1.055) 2.4) (div $gNorm 12.92) -}} {{- $bLum := cond (gt $bNorm 0.03928) (pow (div (add $bNorm 0.055) 1.055) 2.4) (div $bNorm 12.92) -}} {{- $luminance := add (add (mul 0.2126 $rLum) (mul 0.7152 $gLum)) (mul 0.0722 $bLum) -}} {{- /* Choose white for dark backgrounds, black for light backgrounds */ -}} {{- $accentContrastingColor = cond (lt $luminance 0.5) "white" "black" -}} {{- end -}} {{- return dict "accentColor" $accentColor "accentContrastingColor" $accentContrastingColor -}}
