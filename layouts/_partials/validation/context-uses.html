{{- /* Validate that all keys have at least one use and no unrecognised uses */ -}} {{- $contextUsesSection := site.GetPage "/context-uses" -}} {{- $groupsSection := site.GetPage "/groups" -}} {{- if $contextUsesSection -}} {{- /* Build a set of valid context-use names (lowercase for comparison) */ -}} {{- $validContextUses := slice -}} {{- $validContextUsesLower := slice -}} {{- range $contextUsesSection.Pages -}} {{- $contextUseName := .File.TranslationBaseName -}} {{- $validContextUses = $validContextUses | append $contextUseName -}} {{- $validContextUsesLower = $validContextUsesLower | append (lower $contextUseName) -}} {{- end -}} {{- /* Check all keys in all groups */ -}} {{- range $groupsSection.Pages -}} {{- range .Pages -}} {{- $key := .Params.ontology.key -}} {{- $ociAnnotation := .Params.ontology.ociAnnotation -}} {{- $identifier := or $key $ociAnnotation -}} {{- $uses := .Params.ontology.uses -}} {{- if $uses -}} {{- $hasAtLeastOneUse := false -}} {{- range $useName, $useValue := $uses -}} {{- $useNameLower := ($useName | lower) -}} {{- /* Check if this use is recognised (case-insensitive) */ -}} {{- if not (in $validContextUsesLower $useNameLower) -}} {{- errorf "Key '%s' has unrecognised use '%s'. Valid context-uses (case insensitive) are: %v" $identifier $useName $validContextUses -}} {{- end -}} {{- /* Track if at least one use is true */ -}} {{- if (eq $useValue true) -}} {{- $hasAtLeastOneUse = true -}} {{- end -}} {{- end -}} {{- /* Ensure the key has at least one use set to true */ -}} {{- if not $hasAtLeastOneUse -}} {{- errorf "Key '%s' has no uses set to true. At least one use must be enabled" $identifier -}} {{- end -}} {{- else -}} {{- errorf "Key '%s' has no uses defined. Each key must have a uses section" $identifier -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
