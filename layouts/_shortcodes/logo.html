{{- /* Shortcode to render the site logo. Usage: {{< logo >}} - default (white logo) {{< logo scheme="auto" >}} - auto color scheme switching {{< logo scheme="dark" >}} - dark/inverted logo {{< logo scheme="light" >}} - light/white logo {{< logo scheme="auto" class="about-logo" >}} - with custom class */ -}} {{- $scheme := .Get "scheme" | default "" -}} {{- $class := .Get "class" | default "" -}} {{- $id := .Get "id" | default "" -}} {{- partial "logo.html" (dict "scheme" $scheme "class" $class "id" $id) -}}
